function buildSmsAsEmailLayoutJson(row) {
  var tmpl = getTemplateForRow(row);

  // SMS body entered in Study Information only
  var smsBody = String(row["Study Information"] || "");
  smsBody = replaceMergeFields(smsBody, row); // replaces [Add URL] + name brackets, etc.

  // Normalize whitespace a bit (optional, but helps SMS readability)
  smsBody = smsBody
    .replace(/\r\n/g, "\n")
    .replace(/\n{3,}/g, "\n\n")
    .trim();

  var json = {
    "root": {
      "type": "EmailLayout",

        "borderRadius": tmpl.layout.borderRadius,
        "canvasColor": tmpl.layout.canvasColor,

          "fontSize": tmpl.bodyBlock.fontSize,

        "props": {
          // Keep the same paragraph logic + ewa-rteLine you use for email text blocks
          "text": textToDivHtml(smsBody, row)
        }
      }
    }
  };

  return json;

/***********************
 *  HELPERS
 ***********************/
function isSmsRow(row) {
  return /sms/i.test(String(row["Message Name"] || ""));
}

function replaceMergeFields(text, row) {
  if (!text) return "";
  row = row || {};

  var s = String(text);

  // 1) Replace [Add URL] first
  var url = String(row["URL/UTM"] || "").trim();
  if (url) {
    s = s.replace(/\[\s*add\s*url\s*\]/gi, url);
  }

  // 2) Replace any remaining bracketed placeholder with first name
  // (covers [Patient First Name], [nombre del paciente], etc.)
  s = s.replace(/\[\s*(patient\s*first\s*name|nombre\s+del\s+paciente)\s*\]/gi, "{{recipient.firstName}}");


  return s;
}


function buildSmsJson(row) {
  var language = row["Language"] || "en-us";

  // Only use Study Information as body template
  var smsBody = String(row["Study Information"] || "");

  // Apply merge fields + [Add URL] replacement
  smsBody = replaceMergeFields(smsBody, row);

  // Normalize whitespace for SMS readability
  smsBody = smsBody
    .replace(/\r\n/g, "\n")
    .replace(/\n{3,}/g, "\n\n")
    .trim();

  // Minimal SMS payload (adjust keys if your platform expects different)
  return {
    "type": "SmsMessage",
    "data": {
      "language": language,
      "text": smsBody
    }
  };
}


function rowToObject(headers, row) {
  var obj = {};
  headers.forEach(function(h, i) {
    obj[h] = row[i];
  });
  return obj;
}

function getTemplateForRow(row) {
  // Only Walgreens for now; easy to extend later if you have other sponsors
  return EMAIL_TEMPLATES["Walgreens"];
}

function textToDivHtml(text, row) {
  if (!text) return "";

  text = replaceMergeFields(text, row);

  var normalized = String(text).replace(/\r\n/g, "\n");
  var paragraphs = normalized.split(/\n{2,}/);

  return paragraphs
    .map(function(p) { return p.trim(); })
    .filter(function(p) { return p.length > 0; })
    .map(function(p) {

      // Escape HTML (must be first!)
      var escaped = p
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;");

      // Apply bold/italic formatting (markdown-like)
      escaped = applyInlineFormatting(escaped);

      // Convert single line breaks to <br>
      escaped = escaped.replace(/\n/g, "<br/>");

      return '<p class="ewa-rteLine" style="text-align: left;">' + escaped + '</p>';

    })
    .join("");
}

function applyInlineFormatting(text) {
  if (!text) return text;

  // Bold *text*
  text = text.replace(/\*(.+?)\*/g, "<strong>$1</strong>");
  // Italic _text_
  text = text.replace(/_(.+?)_/g, "<em>$1</em>");

  return text;
}

function normalizeImageUrl(url) {
  if (!url) return "";

  var s = String(url).trim();

  // Convert GitHub blob URLs to raw URLs
  // Example:
  // https://github.com/org/repo/blob/main/path/image.jpg
  // → https://raw.githubusercontent.com/org/repo/main/path/image.jpg
  if (s.indexOf("github.com") !== -1 && s.indexOf("/blob/") !== -1) {
    s = s
      .replace("https://github.com/", "https://raw.githubusercontent.com/")
      .replace("/blob/", "/");
  }

  return s;
}


/*************************************
 *  CORE: GENERATE JSON FOR A ROW
 *************************************/
function generateJsonForRow(rowIndex) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();

  var dataRange = sheet.getDataRange();
  var values = dataRange.getValues();
  var headers = values[0];

  if (rowIndex <= 1 || rowIndex > values.length) {
    throw new Error('Row index out of range or header row.');
  }

  var rowValues = values[rowIndex - 1];
  var row = rowToObject(headers, rowValues);

var jsonObject = isSmsRow(row) ? buildSmsAsEmailLayoutJson(row) : buildEmailJson(row);

  var campaignName = row["Campaign Name"] || ("row" + rowIndex);
  var safeName = campaignName.replace(/\s+/g, "");
  var fileName = "initialEmail_" + safeName + ".json";

  var file = DriveApp.createFile(
    fileName,
    JSON.stringify(jsonObject, null, 2),
    MimeType.PLAIN_TEXT
  );

  return file.getUrl(); // so callers can use it (checkbox / menu)
}

/*************************************
 *  MENU ACTION
 *************************************/
function generateJsonForSelectedRow() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();
  var active = sheet.getActiveCell();
  var rowIndex = active.getRow();

  if (rowIndex === 1) {
    SpreadsheetApp.getUi()
      .alert('Please select a data row (not the header row).');
    return;
  }

  try {
    var fileUrl = generateJsonForRow(rowIndex);
    // Put link into column A of that row
    sheet.getRange(rowIndex, 1)
      .setFormula('=HYPERLINK("' + fileUrl + '","JSON")');
    SpreadsheetApp.getUi()
      .alert("JSON generated for row " + rowIndex);
  } catch (err) {
    SpreadsheetApp.getUi()
      .alert("Error: " + err.message);
  }
}

/*************************************
 *  "BUTTON" VIA CHECKBOX IN COLUMN A
 *  (INSTALLABLE on-edit trigger)
 *************************************/
function onEditGenerate_v1_1(e) {
  var range = e.range;
  var sheet = range.getSheet();
  var row = range.getRow();
  var col = range.getColumn();

  // Only react on first column (A) and not header row
  if (col !== 1 || row === 1) return;

  // Optional: limit to one sheet by name
  // if (sheet.getName() !== 'Campaign Email JSON Generator') return;

  var newValue = e.value;

  // Only act when checkbox is checked (TRUE)
  if (newValue === "TRUE") {
    try {
      var fileUrl = generateJsonForRow(row);
      // Replace checkbox with hyperlink
      sheet.getRange(row, col)
        .setFormula('=HYPERLINK("' + fileUrl + '","JSON")');
    } catch (err) {
      SpreadsheetApp.getUi()
        .alert("Error generating JSON for row " + row + ": " + err.message);
      // Reset checkbox if error
      sheet.getRange(row, col).setValue(false);
    }
  }
}
